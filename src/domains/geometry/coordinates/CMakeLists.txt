cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GACoordinateSystem LANGUAGES CXX)
    include(CTest)
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/source.cmake")
add_library(gaCoordinateContracts INTERFACE)
add_library(GeometryArtist::CoordinateContracts ALIAS gaCoordinateContracts)
target_compile_features(gaCoordinateContracts INTERFACE cxx_std_20)
target_include_directories(gaCoordinateContracts INTERFACE
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:include>")
target_sources(gaCoordinateContracts INTERFACE ${GA_COORDINATE_PUBLIC_HEADERS})
add_library(gaCoordinateCore STATIC ${GA_COORDINATE_SOURCES})
add_library(GeometryArtist::CoordinateCore ALIAS gaCoordinateCore)
target_link_libraries(gaCoordinateCore PUBLIC GeometryArtist::CoordinateContracts)
set_target_properties(gaCoordinateCore PROPERTIES POSITION_INDEPENDENT_CODE ON
    CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
    AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(MSVC)
    target_compile_options(gaCoordinateCore PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(gaCoordinateFramesTests ${GA_COORDINATE_TEST_SOURCES})
    target_link_libraries(gaCoordinateFramesTests PRIVATE GeometryArtist::CoordinateCore)
    set_target_properties(gaCoordinateFramesTests PROPERTIES CXX_EXTENSIONS OFF
        AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    if(MSVC)
        target_compile_options(gaCoordinateFramesTests PRIVATE /utf-8)
    endif()
    add_test(NAME ga_coordinate_frames COMMAND gaCoordinateFramesTests)
endif()
