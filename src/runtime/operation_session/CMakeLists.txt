cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(GAOperationSession LANGUAGES CXX)
    include(CTest)
endif()
include(source.cmake)
find_package(Threads REQUIRED)
add_library(gaOperationSessionContracts INTERFACE)
add_library(GeometryArtist::OperationSessionContracts ALIAS gaOperationSessionContracts)
target_compile_features(gaOperationSessionContracts INTERFACE cxx_std_20)
target_include_directories(gaOperationSessionContracts INTERFACE "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_sources(gaOperationSessionContracts INTERFACE ${GA_OPERATION_SESSION_HEADERS})
add_library(gaOperationSessionCore STATIC ${GA_OPERATION_SESSION_SOURCES})
add_library(GeometryArtist::OperationSessionCore ALIAS gaOperationSessionCore)
target_link_libraries(gaOperationSessionCore PUBLIC GeometryArtist::OperationSessionContracts PRIVATE Threads::Threads)
set_target_properties(gaOperationSessionCore PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF
    POSITION_INDEPENDENT_CODE ON AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
if(MSVC)
    target_compile_options(gaOperationSessionCore PRIVATE /utf-8)
endif()
if(BUILD_TESTING)
    add_executable(gaOperationSessionTests ${GA_OPERATION_SESSION_TEST_SOURCES})
    target_link_libraries(gaOperationSessionTests PRIVATE GeometryArtist::OperationSessionCore)
    set_target_properties(gaOperationSessionTests PROPERTIES CXX_EXTENSIONS OFF AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    add_test(NAME gaOperationSessionCore COMMAND gaOperationSessionTests)
    set_tests_properties(gaOperationSessionCore PROPERTIES TIMEOUT 30)
endif()
